// 安裝依賴：npm install @capacitor/haptics import { Haptics, ImpactStyle } from '@capacitor/haptics'; const HapticBreathingDriver = { hapticTimer: null, // 吸氣：4 秒內頻率由慢漸快、強度遞增 startInhale() { this.stop(); let step = 0; this.hapticTimer = setInterval(async () => { if (step % 2 === 0) { await Haptics.impact({ style: ImpactStyle.Light }).catch(() => {}); } step++; }, 400 - (step * 25)); // 間隔逐漸縮短 }, // 屏息：模擬微弱安靜的心跳雙脈衝 (Lubb-Dupp) startHold() { this.stop(); this.hapticTimer = setInterval(async () => { await Haptics.impact({ style: ImpactStyle.Medium }).catch(() => {}); setTimeout(async () => { await Haptics.impact({ style: ImpactStyle.Light }).catch(() => {}); }, 120); }, 1200); }, // 呼氣：8 秒平緩釋放，只保留極微弱提示 startExhale() { this.stop(); let count = 0; this.hapticTimer = setInterval(async () => { if (count < 4) { await Haptics.impact({ style: ImpactStyle.Light }).catch(() => {}); } count++; }, 1500); }, stop() { if (this.hapticTimer) clearInterval(this.hapticTimer); this.hapticTimer = null; } };
