// ========== 密碼雜湊 ==========

async function hashPassword(password, salt) {

    const enc = new TextEncoder();

    const key = await crypto.subtle.importKey('raw', enc.encode(password), 'PBKDF2', false, ['deriveBits']);

    const bits = await crypto.subtle.deriveBits(

        { name: 'PBKDF2', salt: enc.encode(salt), iterations: 100000, hash: 'SHA-256' },

        key, 256

    );

    return btoa(String.fromCharCode(...new Uint8Array(bits)));

}

function getWeekKey() {

    monday.setDate(today.getDate() - ((today.getDay() + 6) % 7));

}

function calculateCalmScore() {

    const streakScore = Math.min(40, appState.progress.streak * 4);

    const weekMinutes = appState.progress.weeklyData.reduce((a, b) => a + b, 0);

    const minutesScore = Math.min(30, weekMinutes);

    const moodMap = { '😌': 100, '😊': 85, '😐': 60, '😔': 40, '😰': 20 };

    const recentMoods = appState.progress.moodLog.slice(-7);

    const moodScore = recentMoods.length

        ? (recentMoods.reduce((a, m) => a + (moodMap[m.mood] || 60), 0) / recentMoods.length) * 0.3

        : 15;

    return Math.round(Math.min(100, streakScore + minutesScore + moodScore));

}

    ctx.fillStyle = "#f3f4f6";

    ctx.fillRect(0, 0, canvas.width, canvas.height);

    for (let i = 0; i < 4; i++) {

        ctx.strokeStyle = `rgba(${Math.random()*255}, ${Math.random()*255}, ${Math.random()*255}, 0.3)`;

        ctx.beginPath();

        ctx.moveTo(Math.random() * canvas.width, Math.random() * canvas.height);

        ctx.lineTo(Math.random() * canvas.width, Math.random() * canvas.height);

        ctx.stroke();

    }

    for (let i = 0; i < code.length; i++) {

        ctx.save();

        ctx.font = "bold 22px Courier New";

        ctx.fillStyle = `rgba(${Math.random()*140}, ${Math.random()*140}, ${Math.random()*140}, 0.9)`;

        ctx.rotate((Math.random() * 40 - 20) * Math.PI / 180);

        ctx.restore();

    }

async function submitAuthForm() {

    const userIn = document.getElementById('authUsername').value.trim();

    const passIn = document.getElementById('authPassword').value.trim();

    const captchaIn = document.getElementById('authCaptchaInput').value.trim().toLowerCase();


    if (!userIn || !passIn || !captchaIn) { showToast("請填寫完整欄位"); return; }

    if (captchaIn !== appState.auth.captchaCode) { showToast("驗證碼錯誤"); generateCaptcha(); return; }

    if (passIn.length < 6) { showToast("密碼至少 6 個字元"); return; }


    const hashedKey = await hashPassword(passIn, userIn.toLowerCase());


    if (appState.auth.isSignUpMode) {

        const checkCloud = await fetchMongoDB('findOne', { filter: { username: userIn } });

        if (checkCloud && checkCloud.document) {

            showToast("此用戶名已被註冊"); generateCaptcha(); return;

        }

        appState.auth.currentUser = userIn;

        appState.auth.isLoggedIn = true;

        await fetchMongoDB('insertOne', {

            document: {

                username: userIn,

                passwordHash: hashedKey,

                createdAt: new Date().toISOString(),

                progress: appState.progress,

                settings: appState.settings

            }

        });

        showToast(`註冊成功！歡迎加入，${userIn}`);

    } else {

        const result = await fetchMongoDB('findOne', { filter: { username: userIn } });

        if (result && result.document) {

            if (result.document.passwordHash !== hashedKey) {

                showToast("密碼錯誤"); generateCaptcha(); return;

            }

            if (result.document.progress) {

                appState.progress = { ...appState.progress, ...result.document.progress };

                appState.progress.mindfulDates = appState.progress.mindfulDates || [];

                appState.progress.moodLog = appState.progress.moodLog || [];

            }

        } else if (MONGODB_CONFIG.apiKey !== "YOUR_ATLAS_DATA_API_KEY") {

            showToast("用戶不存在，請先註冊"); generateCaptcha(); return;

        }

        appState.auth.currentUser = userIn;

        appState.auth.isLoggedIn = true;

        await loadUserStateFromStorage(userIn);

    }

    await syncWithCloud('user_login');

    document.getElementById('authUsername').value = '';

    document.getElementById('authPassword').value = '';

    document.getElementById('authCaptchaInput').value = '';

function initAudioEngine() {

    audioEngine.preload = 'auto';

    audioEngine.addEventListener('error', () => {

    audioEngine.addEventListener('loadedmetadata', () => {

        if (Number.isFinite(audioEngine.duration) && audioEngine.duration > 0) {

            appState.session.totalDuration = Math.floor(audioEngine.duration);

            appState.session.isAudioLoaded = true;

        }

        syncDurationUI();

        updateMediaSessionPositionState();

    });

    audioEngine.addEventListener('timeupdate', () => {

        if (appState.session.isAudioLoaded) {

            appState.session.currentTime = audioEngine.currentTime;

            updatePlaybackProgressUI();

        }

    });

    audioEngine.addEventListener('play', () => { handlePlayStartState(); });

    audioEngine.addEventListener('pause', () => { handlePlayPauseState(); });

    audioEngine.addEventListener('ended', () => { handlePlayEndState(); });

}

    timerDisplay.innerText = formatMinutesSeconds(timeLeft);

function handlePlayStartState() {

    appState.session.isPlaying = true;

    playBtn.innerText = "⏸️";

    if ('mediaSession' in navigator) { navigator.mediaSession.playbackState = "playing"; }

    if (backupTimer) clearInterval(backupTimer);

    appState.session.endTime = Date.now() + ((appState.session.totalDuration - appState.session.currentTime) * 1000);

    backupTimer = setInterval(() => {

        if (appState.session.isPlaying) {

            if (!appState.session.isAudioLoaded) {

                const remaining = Math.max(0, Math.floor((appState.session.endTime - Date.now()) / 1000));

                appState.session.currentTime = appState.session.totalDuration - remaining;

                updatePlaybackProgressUI();

                if (remaining === 0) handlePlayEndState();

            } else {

                appState.session.endTime = Date.now() + ((appState.session.totalDuration - audioEngine.currentTime) * 1000);

            }

        }

    }, 250);

}

    if (backupTimer) clearInterval(backupTimer);

    if ('mediaSession' in navigator) { navigator.mediaSession.playbackState = "paused"; }

    saveUserStateToStorage();

}


function handlePlayEndState() {

    appState.session.isPlaying = false;

    playBtn.innerText = "▶️";

    if (backupTimer) clearInterval(backupTimer);

    const minutesDone = Math.max(1, Math.floor(appState.session.totalDuration / 60));

    recordSession(minutesDone);

    syncWithCloud('save_progress', {

        trackId: appState.session.currentTrackId,

        durationMinutes: minutesDone

    });

    appState.session.currentTime = 0;

    updatePlaybackProgressUI();

    saveUserStateToStorage();

    document.getElementById('moodModal').classList.add('active');

}

// ========== 呼吸模式 ==========

function selectPattern(patternKey) {

    appState.session.breathingPattern = patternKey;

    document.querySelectorAll('.pattern-chip').forEach(el => {

        el.classList.toggle('active', el.dataset.pattern === patternKey);

    });

    if (appState.session.isPlaying) {

        runBreathingCycle();

    }

}


function runBreathingCycle() {

    if (breathPhaseTimeout) clearTimeout(breathPhaseTimeout);

    if (!appState.session.isPlaying) return;


    const pattern = BREATHING_PATTERNS[appState.session.breathingPattern];

    const phases = [

        { name: 'Inhale', duration: pattern.inhale, scale: 1.23, vibrate: [35] },

        { name: 'Hold',   duration: pattern.hold,   scale: 1.23, vibrate: [15] },

        { name: 'Exhale', duration: pattern.exhale, scale: 1.0,  vibrate: [20, 60, 20] },

        { name: 'Hold',   duration: pattern.hold2,  scale: 1.0,  vibrate: [15] }

    ].filter(p => p.duration > 0);


    let idx = 0;

    function step() {

        if (!appState.session.isPlaying) return;

        const phase = phases[idx];

        breathTxt.innerText = phase.name + ' ' + (phase.duration / 1000) + 's';

        breathRing.style.transform = `scale(${phase.scale})`;

        if ('vibrate' in navigator && phase.vibrate) navigator.vibrate(phase.vibrate);

        idx = (idx + 1) % phases.length;

        breathPhaseTimeout = setTimeout(step, phase.duration);

    }

    step();

}


function stopBreathingCycle() {

    if (breathPhaseTimeout) clearTimeout(breathPhaseTimeout);

    breathRing.style.transform = 'scale(1.0)';

    if (appState.session.currentTime === 0) breathTxt.innerText = "Ready";

}


// ========== 播放控制 ==========

async function selectTrack(trackId) {

    appState.session.currentTrackId = trackId;

    setupTrackMetadataUI(trackId);

    audioEngine.volume = appState.settings.volume || 0.8;

    audioEngine.src = appState.session.tracks[trackId].url;

    audioEngine.load();

    switchPage('session');

    await togglePlayback();

}


async function togglePlayback() {

    try {

        if (audioEngine.paused) {

            await audioEngine.play();

            handlePlayStartState();

            runBreathingCycle();

        } else {

            audioEngine.pause();

            handlePlayPauseState();

            stopBreathingCycle();

        }

    } catch (err) {

        if (!appState.session.isPlaying) {

            handlePlayStartState();

            runBreathingCycle();

        } else {

            handlePlayPauseState();

            stopBreathingCycle();

        }

    }

}


function handleSeek(event) {

    const validDuration = appState.session.totalDuration;

    if (!validDuration) return;

    const rect = document.getElementById('progressTrack').getBoundingClientRect();

    const targetTime = Math.min(1, Math.max(0, (event.clientX - rect.left) / rect.width)) * validDuration;

    appState.session.currentTime = targetTime;

    if (appState.session.isAudioLoaded) {

        try { audioEngine.currentTime = targetTime; } catch (e) {}

    }

    appState.session.endTime = Date.now() + ((validDuration - targetTime) * 1000);

    updatePlaybackProgressUI();

    updateMediaSessionPositionState();

}


function handleVolumeChange(val) {

    audioEngine.volume = parseFloat(val);

    appState.settings.volume = parseFloat(val);

    saveUserStateToStorage();

}


// ========== MediaSession ==========

function updateMediaSessionMetadata(trackId) {

    if ('mediaSession' in navigator) {

        const track = appState.session.tracks[trackId];

        navigator.mediaSession.metadata = new MediaMetadata({

            title: track.title,

            artist: "Mindful Space",

            album: track.subtitle,

            artwork: [{ src: './icon-192.png', sizes: '192x192', type: 'image/png' }]

        });

    }

}

function updateMediaSessionPositionState() {

    if ('mediaSession' in navigator && 'setPositionState' in navigator.mediaSession) {

        if (Number.isFinite(audioEngine.duration) && audioEngine.duration > 0) {

            navigator.mediaSession.setPositionState({

                duration: audioEngine.duration,

                playbackRate: audioEngine.playbackRate,

                position: audioEngine.currentTime

            });

        }

    }

}

function initMediaSessionActionHandlers() {

    if ('mediaSession' in navigator) {

        navigator.mediaSession.setActionHandler('play', () => togglePlayback());

        navigator.mediaSession.setActionHandler('pause', () => togglePlayback());

    }

}

}
